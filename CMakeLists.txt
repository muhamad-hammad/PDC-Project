cmake_minimum_required(VERSION 3.16)
project(unary_state_machines LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED)
add_executable(unary src/main.cpp)
target_link_libraries(unary PRIVATE OpenMP::OpenMP_CXX)
if(MSVC)
  target_compile_options(unary PRIVATE /W4)
else()
  target_compile_options(unary PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME correctness COMMAND unary test)
add_test(NAME show_add COMMAND unary show add 2 3)
set_tests_properties(show_add PROPERTIES PASS_REGULAR_EXPRESSION "result=5")
add_test(NAME show_sub COMMAND unary show sub 2 3)
set_tests_properties(show_sub PROPERTIES PASS_REGULAR_EXPRESSION "result=0")
add_test(NAME show_mul COMMAND unary show mul 3 4)
set_tests_properties(show_mul PROPERTIES PASS_REGULAR_EXPRESSION "result=12")
add_test(NAME show_div COMMAND unary show div 7 3)
set_tests_properties(show_div PROPERTIES PASS_REGULAR_EXPRESSION "result=2")
add_test(NAME show_mod COMMAND unary show mod 7 3)
set_tests_properties(show_mod PROPERTIES PASS_REGULAR_EXPRESSION "result=1")
